pub fn serve(st: &State, media: &Media, ws: WebSocket) -> Result<(), String> {

    send(&mut ws, json!({"t": "chal", "c": chal}))?;
    let (id, mac) = hello(&mut ws, HELLO_DEADLINE)?;
    if !st.devices.admit(&id, &chal, &mac) {

    let fp_mac = st.devices.fp_mac(&id, &chal, &media.fp);

    let answer = rtc::prepare(&offer, local_ip, media.cert.clone())?;
